## 7. 算法参考实现与坐标转换接口

```python
"""
周天六十极轨 (HexaCelestial Grid) Python 算法验证模块
"""
from typing import Tuple, Dict

# 五行与天层定义 (赤纬区间)
LAYERS = [
    {"name": "北水", "stem_yang": "壬", "stem_yin": "癸", "dec_min": 66.5, "dec_max": 90.0},
    {"name": "次北木", "stem_yang": "甲", "stem_yin": "乙", "dec_min": 23.5, "dec_max": 66.5},
    {"name": "中土", "stem_yang": "戊", "stem_yin": "己", "dec_min": -23.5, "dec_max": 23.5},
    {"name": "次南金", "stem_yang": "庚", "stem_yin": "辛", "dec_min": -66.5, "dec_max": -23.5},
    {"name": "南火", "stem_yang": "丙", "stem_yin": "丁", "dec_min": -90.0, "dec_max": -66.5},
]

BRANCHES = ["子", "丑", "寅", "卯", "辰", "巳", "午", "未", "申", "酉", "戌", "亥"]

def ecliptic_to_grid(lon: float, lat_dec: float) -> str:
    """
    将连续天球坐标转换为四字干支码
    lon: 黄经 0.0 ~ 360.0 度
    lat_dec: 赤纬 -90.0 ~ +90.0 度
    """
    lon = lon % 360.0
    lat_dec = max(-90.0, min(90.0, lat_dec))
    
    # 1. 解算第一阶宏观天层 (5 带)
    layer_idx = 0
    if lat_dec >= 66.5:
        layer_idx = 0  # 北水
    elif lat_dec >= 23.5:
        layer_idx = 1  # 次北木
    elif lat_dec >= -23.5:
        layer_idx = 2  # 中土
    elif lat_dec >= -66.5:
        layer_idx = 3  # 次南金
    else:
        layer_idx = 4  # 南火
        
    layer = LAYERS[layer_idx]
    
    # 2. 解算第一阶宏观黄道辰 (12 辰)
    branch_idx = int(lon // 30)
    branch_char = BRANCHES[branch_idx]
    
    # 依据地支奇偶选择天干阴阳，严格维持同奇偶律
    if branch_idx % 2 == 0:
        stem_char = layer["stem_yang"]
    else:
        stem_char = layer["stem_yin"]
        
    macro_code = f"{stem_char}{branch_char}"
    
    # 3. 解算第二阶微观网格 (大区内部 60 细分)
    # 黄经剩余微角
    rem_lon = lon - (branch_idx * 30.0)  # 0.0 ~ 30.0
    micro_branch_idx = int(rem_lon // 2.5)  # 0 ~ 11
    micro_branch_char = BRANCHES[micro_branch_idx]
    
    # 赤纬在带内的归一化占比
    l_min = layer["dec_min"]
    l_max = layer["dec_max"]
    norm_lat = (lat_dec - l_min) / (l_max - l_min)  # 0.0 ~ 1.0
    micro_layer_idx = min(4, int(norm_lat * 5))
    micro_layer = LAYERS[micro_layer_idx]
    
    if micro_branch_idx % 2 == 0:
        micro_stem_char = micro_layer["stem_yang"]
    else:
        micro_stem_char = micro_layer["stem_yin"]
        
    micro_code = f"{micro_stem_char}{micro_branch_char}"
    
    return f"{macro_code} · {micro_code}"
```
